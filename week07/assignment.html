Part 2: Sex-associated motif accessibility differences

data_dir <- if (basename(getwd()) == "week07") "." else "week07"
peak_file <- file.path(data_dir, "peakCounts.rds")

if (!file.exists(peak_file)) {
  download.file(
    "https://ethz-ins.org/content/mouse_mm38_hippocampus.peakCounts.SE.rds",
    peak_file,
    mode = "wb"
  )
}

se <- readRDS(peak_file)
se$sex <- ifelse(grepl("_female_", colnames(se)), "Female", "Male")
se$group <- ifelse(grepl("^FSS_", colnames(se)), "Stressed", "Control")
table(se$group, se$sex)
##           
##            Female Male
##   Control       3    3
##   Stressed      3    3
se <- GenomeInfoDb::keepStandardChromosomes(se, pruning.mode = "coarse")
GenomeInfoDb::seqlevelsStyle(se) <- "UCSC"
se <- betterChromVAR::addGCBias(se, genome = BSgenome.Mmusculus.UCSC.mm10)
"bias" %in% names(SummarizedExperiment::rowData(se))
## [1] TRUE
motifs <- MotifDb::query(MotifDb, "HOCOMOCOv13")

motifs <- do.call(
  TFBSTools::PWMatrixList,
  universalmotif::convert_motifs(motifs, class = "TFBSTools-PWMatrix")
)

names(motifs) <- make.unique(sapply(motifs, \(x) gsub("\\..+", "", x@name)))

matches <- motifmatchr::matchMotifs(
  motifs,
  se,
  genome = BSgenome.Mmusculus.UCSC.mm10
)

design <- model.matrix(
  ~ group + sex,
  data = as.data.frame(SummarizedExperiment::colData(dev))
)

fit <- limma::eBayes(
  limma::lmFit(SummarizedExperiment::assay(dev, "z"), design)
)

res <- as.data.frame(
  limma::topTable(fit, coef = "sexMale", number = Inf)
)
