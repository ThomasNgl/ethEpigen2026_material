find_col <- function(nms, choices, pattern = NULL) {
  hit <- match(tolower(choices), tolower(nms), nomatch = 0)
  if (any(hit > 0)) return(nms[hit[hit > 0][1]])

  if (!is.null(pattern)) {
    hit <- grep(pattern, nms, ignore.case = TRUE, value = TRUE)
    if (length(hit) > 0) return(hit[1])
  }

  NA_character_
}

peak_info <- data.frame(
  peak_id = peaks$peak_id,
  peak_chr = as.character(seqnames(peaks)),
  peak_center = start(peaks) + floor((width(peaks) - 1) / 2)
)

tss_info <- data.frame(
  gene_name = tss$gene_name,
  peak_chr = as.character(seqnames(tss)),
  target_tss = start(tss),
  target_strand = as.character(strand(tss))
) |>
  distinct()

analysis_tbl <- peak_targets |>
  mutate(link_id = row_number()) |>
  left_join(peak_info, by = "peak_id") |>
  inner_join(
    tss_info,
    by = c("target_gene" = "gene_name", "peak_chr"),
    relationship = "many-to-many"
  ) |>
  mutate(
    signed_distance = ifelse(
      target_strand == "+",
      peak_center - target_tss,
      target_tss - peak_center
    ),
    target_distance = abs(signed_distance),
    direction = ifelse(signed_distance < 0, "upstream", "downstream")
  ) |>
  group_by(link_id) |>
  slice_min(target_distance, n = 1, with_ties = FALSE) |>
  ungroup() |>
  left_join(nearest_tbl, by = "peak_id") |>
  select(
    peak_id, target_gene, target_tss, target_strand,
    signed_distance, target_distance, direction,
    nearest_gene, nearest_tss_distance
  )

I used GATA2 ChIP-seq peaks from K562 cells and ENCODE-rE2G thresholded enhancer-gene predictions from the same cell type. Of 15650 total GATA2 peaks, 8283 overlapped at least one rE2G element-gene link and were therefore kept for the comparison. For upstream peak-target links, the predicted rE2G target was also the nearest TSS gene in 34.8% of cases. For downstream peak-target links, this was true in 37.9% of cases. The median distance between the peak center and the predicted target TSS was 27.3 kb for upstream links and 27.1 kb for downstream links. These results show that rE2G assigns many GATA2-bound regulatory regions to predicted target genes that are not necessarily the nearest gene in linear genomic distance.
